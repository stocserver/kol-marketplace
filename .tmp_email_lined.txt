3: type Recipient = { email: string; name?: string }
4: 
5: interface SendEmailOptions {
6:   to: Recipient[]
7:   subject: string
8:   text?: string
9:   html?: string
10:   category?: string
11:   fromEmail?: string
12:   fromName?: string
13: }
14: 
15: const MAILTRAP_ENDPOINT = 'https://send.api.mailtrap.io/api/send'
16: 
17: function getEnv(name: string) {
18:   const v = process.env[name]
19:   return typeof v === 'string' && v.trim().length > 0 ? v : undefined
20: }
21: 
22: export function isEmailConfigured() {
23:   return Boolean(getEnv('MAILTRAP_TOKEN') && (getEnv('MAILTRAP_FROM_EMAIL')))
24: }
25: 
26: export async function sendEmail(options: SendEmailOptions) {
27:   const token = getEnv('MAILTRAP_TOKEN')
28:   const fromEmail = options.fromEmail || getEnv('MAILTRAP_FROM_EMAIL')
29:   const fromName = options.fromName || getEnv('MAILTRAP_FROM_NAME') || 'Kol Marketplace'
30:   const category = options.category || getEnv('MAILTRAP_CATEGORY') || 'Transactional'
31: 
32:   if (!token || !fromEmail) {
33:     throw new Error('Email not configured. Set MAILTRAP_TOKEN and MAILTRAP_FROM_EMAIL')
34:   }
35: 
36:   const payload = {
37:     from: { email: fromEmail, name: fromName },
38:     to: options.to.map((r) => ({ email: r.email, name: r.name })),
39:     subject: options.subject,
40:     text: options.text,
41:     html: options.html,
42:     category,
43:   }
44: 
45:   const res = await fetch(MAILTRAP_ENDPOINT, {
46:     method: 'POST',
47:     headers: {
48:       Authorization: `Bearer ${token}`,
49:       'Content-Type': 'application/json',
50:     },
51:     body: JSON.stringify(payload),
52:     // Mail must be sent from server runtime only
53:     cache: 'no-store',
54:   })
55: 
56:   if (!res.ok) {
57:     const body = await res.text().catch(() => '')
58:     throw new Error(`Mailtrap send failed: ${res.status} ${res.statusText} ${body}`)
59:   }
60: 
61:   try {
62:     return await res.json()
63:   } catch {
64:     return { ok: true }
65:   }
66: }

79: export async function sendOrderReceivedEmail(params: OrderEmailParams) {
80:   const to: Recipient = typeof params.to === 'string' ? { email: params.to } : params.to
81:   const currency = params.currency || 'USD'
82: 
83:   const subject = `New Order Received #${params.orderId}`
84:   const text = [
85:     `You received a new order ${params.orderId}.`,
86:     params.gigTitle ? `Gig: ${params.gigTitle}` : undefined,
87:     params.buyerName ? `Buyer: ${params.buyerName}` : undefined,
88:     `Amount: ${params.amount} ${currency}`,
89:     params.orderUrl ? `View Order: ${params.orderUrl}` : undefined,
90:   ]
91:     .filter(Boolean)
92:     .join('\n')
93: 
94:   const html = `
95:   <div style="font-family: Arial, sans-serif; line-height: 1.6; color: #111">
96:     <h2 style="margin:0 0 12px">You received a new order</h2>
97:     <p style="margin:0 0 16px; color:#444">${params.kolName ? `${params.kolName}, ` : ''}great news — a buyer just placed an order.</p>
98:     <table style="border-collapse: collapse; width: 100%; max-width: 520px">
99:       <tr>
100:         <td style="padding:8px; border:1px solid #eee; font-weight:600">Order ID</td>
101:         <td style="padding:8px; border:1px solid #eee">${params.orderId}</td>
102:       </tr>
103:       ${params.gigTitle ? `<tr><td style="padding:8px; border:1px solid #eee; font-weight:600">Gig</td><td style="padding:8px; border:1px solid #eee">${params.gigTitle}</td></tr>` : ''}
104:       ${params.buyerName ? `<tr><td style="padding:8px; border:1px solid #eee; font-weight:600">Buyer</td><td style="padding:8px; border:1px solid #eee">${params.buyerName}</td></tr>` : ''}
105:       <tr>
106:         <td style="padding:8px; border:1px solid #eee; font-weight:600">Amount</td>
107:         <td style="padding:8px; border:1px solid #eee">${params.amount} ${currency}</td>
108:       </tr>
109:     </table>
110:     ${
111:       params.orderUrl
112:         ? `<p style="margin:16px 0"><a href="${params.orderUrl}" style="background:#2563eb;color:#fff;padding:10px 16px;border-radius:6px;text-decoration:none;display:inline-block">View Order</a></p>`
113:         : ''
114:     }
115:     <p style="margin:16px 0; color:#666; font-size:13px">This is an automated notification from Kol Marketplace.</p>
116:   </div>`

118:   return sendEmail({
119:     to: [to],
120:     subject,
121:     text,
122:     html,
123:     category: 'Order Notifications',
124:   })
125: }
